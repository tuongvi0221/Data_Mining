from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
import seaborn as sns
import matplotlib.pyplot as plt

# Huấn luyện mô hình phân lớp trên tập train
clf = DecisionTreeClassifier(random_state=42)
clf.fit(X_train, y_train)

# Dự đoán nhãn lớp cho tập test
y_pred = clf.predict(X_test)

# Tính và in ra Accuracy của mô hình trên tập test
accuracy = accuracy_score(y_test, y_pred)
print(f"Accuracy của mô hình trên tập test: {accuracy:.4f}")

# Tính và in ra Precision, Recall, F1-score của từng lớp và trung bình của mô hình trên tập test
print("\nPrecision, Recall, F1-score:")
print(classification_report(y_test, y_pred))

# Tính và hiển thị Confusion Matrix
cm = confusion_matrix(y_test, y_pred)
plt.figure(figsize=(8, 6))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', xticklabels=iris.target_names, yticklabels=iris.target_names)
plt.title("Confusion Matrix")
plt.xlabel("Predicted Labels")
plt.ylabel("True Labels")
plt.show()

import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
from tensorflow.keras.utils import to_categorical
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
import seaborn as sns
import matplotlib.pyplot as plt

# Chuyển đổi y_train và y_test sang dạng one-hot encoding
y_train_onehot = to_categorical(y_train, num_classes=3)  # 3 lớp trong bộ dữ liệu Iris
y_test_onehot = to_categorical(y_test, num_classes=3)

# Xây dựng mô hình Neural Network
model = Sequential([
    Dense(10, activation='relu', input_shape=(X_train.shape[1],)),  # Hidden layer 1
    Dense(20, activation='relu'),  # Hidden layer 2
    Dense(3, activation='softmax')  # Output layer
])

# Biên dịch mô hình
model.compile(optimizer='adam',
              loss='sparse_categorical_crossentropy',  # Sử dụng sparse_categorical_crossentropy cho nhãn dạng số nguyên
              metrics=['accuracy'])

# Huấn luyện mô hình
history = model.fit(X_train, y_train, epochs=100, batch_size=1, verbose=1)

# Dự đoán nhãn lớp cho tập test
y_pred_probs = model.predict(X_test)
y_pred_nn = np.argmax(y_pred_probs, axis=1) #Lấy ra xác suất có giá trị cao nhất, thứ tự của nó sẽ gán cho y dự đoán

# Tính và in Accuracy
accuracy_nn = accuracy_score(y_test, y_pred_nn)
print(f"Accuracy của Neural Network trên tập test: {accuracy_nn:.4f}")

# Tính và in Precision, Recall, F1-score
print("\nPrecision, Recall, F1-score cho Neural Network:")
print(classification_report(y_test, y_pred_nn))

# Tính và hiển thị Confusion Matrix
cm_nn = confusion_matrix(y_test, y_pred_nn)
plt.figure(figsize=(8, 6))
sns.heatmap(cm_nn, annot=True, fmt='d', cmap='Blues', xticklabels=iris.target_names, yticklabels=iris.target_names)
plt.title("Confusion Matrix - Neural Network")

plt.ylabel("True Labels")

    [7.3, 2.8, 6.4, 2.1]
])

# Chuẩn hóa dữ liệu mới tương tự như X_train

new_samples_normalized = new_samples / np.max(iris.data)

print("Dự đoán nhãn lớp cho các mẫu mới:\n")

# Câu e: Decision Tree
dt_predictions = clf.predict(new_samples_normalized)
print(f"Decision Tree dự đoán: {dt_predictions}")

# Câu f: Naïve Bayes
nb_predictions = nb_model.predict(new_samples_normalized)
print(f"Naïve Bayes dự đoán: {nb_predictions}")

# Câu g: K Nearest Neighbors
knn_predictions = knn_model.predict(new_samples_normalized)
print(f"KNN dự đoán: {knn_predictions}")

# Câu h: Neural Network
nn_probabilities = model.predict(new_samples_normalized)  # Lấy xác suất dự đoán từ mô hình Neural Network

print(f"Neural Network dự đoán: {nn_predictions}")
