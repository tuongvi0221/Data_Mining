In [5]:
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Flatten, Dense, Dropout

# Xây dựng mô hình
model = Sequential([
    # Flatten layer: Chuyển từ 28x28 thành 784
    Flatten(input_shape=(28, 28)),

    # Dense layer: 128 neurons với hàm kích hoạt Relu
    Dense(128, activation='relu'),

    # Dropout layer: 20% các neurons bị tắt
    Dropout(0.2),

    # Dense layer: 10 neurons (số class), hàm kích hoạt Softmax
    Dense(10, activation='softmax')
])

# Hiển thị kiến trúc mô hình
model.summary()
