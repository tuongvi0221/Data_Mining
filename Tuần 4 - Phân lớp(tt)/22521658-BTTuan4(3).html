# d) Hàm tạo mô hình
def create_model():
    model = Sequential([
        Flatten(input_shape=(28, 28)),  # Flatten ảnh 28x28 thành vector 784
        Dense(128, activation='relu'),
        Dropout(0.2),
        Dense(10, activation='softmax')  # Output layer với 10 nhãn
    ])
    model.compile(optimizer=Adam(),
                  loss='sparse_categorical_crossentropy',
                  metrics=['accuracy'])
    return model

Huấn luyện mô hình bằng k-fold cross validation (với k=10).

In [6]:
kfold = KFold(n_splits=10, shuffle=True, random_state=42)
# Mảng lưu kết quả
models = []  # Lưu các mô hình
accuracy_per_fold = []  # Lưu accuracy của từng fold
loss_per_fold = []  # Lưu loss của từng fold

# Lặp qua mỗi fold
fold_no = 1
for train_idx, val_idx in kfold.split(x_train, y_train):
    print(f"Đang huấn luyện fold {fold_no}...")

    # Tạo tập train và validation cho fold này
    X_train_fold, X_val_fold = x_train[train_idx], x_train[val_idx]
    y_train_fold, y_val_fold = y_train[train_idx], y_train[val_idx]

    # Tạo mô hình
    model = create_model()

    # Huấn luyện mô hình
    history = model.fit(X_train_fold, y_train_fold,
                        validation_data=(X_val_fold, y_val_fold),
                        epochs=5, batch_size=32, verbose=0)

    # Lưu mô hình vào mảng models
    models.append(model)

    # Tính và in accuracy và loss cho fold này
    loss, accuracy = model.evaluate(X_val_fold, y_val_fold, verbose=0)
    accuracy_per_fold.append(accuracy)
    loss_per_fold.append(loss)

    print(f"Fold {fold_no} - Accuracy: {accuracy:.4f} - Loss: {loss:.4f}")

    fold_no += 1

# In kết quả tổng hợp
print("\nKết quả tổng hợp:")
for i in range(10):
    print(f"Fold {i+1} - Accuracy: {accuracy_per_fold[i]:.4f} - Loss: {loss_per_fold[i]:.4f}")

import matplotlib.pyplot as plt

# Vẽ biểu đồ Accuracy và Loss
def plot_accuracy_loss(accuracy_per_fold, loss_per_fold):
    folds = range(1, len(accuracy_per_fold) + 1)

    plt.figure(figsize=(12, 6))

    # Accuracy
    plt.subplot(1, 2, 1)
    plt.plot(folds, accuracy_per_fold, marker='o', linestyle='-', color='b')
    plt.title('Accuracy per Fold')
    plt.xlabel('Fold')
    plt.ylabel('Accuracy')
    plt.xticks(folds)

    # Loss
    plt.subplot(1, 2, 2)
    plt.plot(folds, loss_per_fold, marker='o', linestyle='-', color='r')
    plt.title('Loss per Fold')
    plt.xlabel('Fold')
    plt.ylabel('Loss')
    plt.xticks(folds)

    plt.tight_layout()
    plt.show()

# Gọi hàm để vẽ biểu đồ
plot_accuracy_loss(accuracy_per_fold, loss_per_fold)
